    ./rjtd/target/debug/rjtd page-layer-tree "$f" 0 |
      jq -c '.root.ops[]? | select(.pageSpaceSolver?) | .pageSpaceSolver.sourcePageYTransformGate.sourceOnlyPageYRenderAdmissionGate // empty' |
      head -n 1
  )
  if [ -z "$gate" ]; then
    printf '%s\tNO_GATE\n' "$name"

    printf '%s\t' "$name"
    printf '%s' "$gate" |
      jq -r '[
        (.admissionReady|tostring),

        (.pageOriginAuthority // "null"),
        (.lineMarkRowsExactAndContiguous|tostring),
        (.lineMarkPageOriginPresent|tostring),
        (.lineMarkPageOriginStridePresent|tostring),
        (.sourceOnlySelectorBlockedReason // "null"),
        ((.blockedReasons // []) | join(","))
      ] | @tsv'
  fi
done
